-----5.增量文件中的额度高于我行授信额度

---借呗
union

select 'ALIPAY' as proj_no,'增量文件中的额度高于我行授信额度' as rule,a.cert_no, '${cdate}'  from
(
select common_string_1 as cert_no,cast(common_string_4 as decimal(16,2))/100 as credit_quota
from net_ods_alipay_info_change_di where cdate = '${cdate}' and change_type = 'TOTAL_CREDIT_AMT'
)a
left join
(
select cert_no,credit_quota from
(
select cert_no,credit_quota,row_number() over (partition by cert_no order by credit_date desc) as rn
from net_dw_loan_apply_valid where proj_no in ('ALIPAY','XZD_ALIPAY') and quota_status in (1,2)  and credit_date <= '${cdate}'
)tt
where rn = 1
)b
on a.cert_no = b.cert_no
where a.credit_quota > b.credit_quota

union

---花呗

select 'HUABEI' as proj_no,'增量文件中的额度高于我行授信额度' as rule,a.cert_no, '${cdate}' from
(

from net_ods_huabei_bill_daily_change_memo_di where cdate = '${cdate}' and change_type = 'TOTAL_CREDIT_AMT'
)a
left join
(
select cert_no,credit_quota from
(
select cert_no,credit_quota,row_number() over (partition by cert_no order by credit_date desc) as rn

)tt
where rn = 1
)b
on a.cert_no = b.cert_no
where a.credit_quota > b.credit_quota
