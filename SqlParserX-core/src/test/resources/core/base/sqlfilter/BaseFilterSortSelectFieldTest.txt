SELECT
    o.*,
    p.name,
    p.description
FROM
    (SELECT
        *
     FROM
        orders
     WHERE
        order_status = FALSE
    ) AS o
LEFT JOIN products AS p ON o.product_id = p.id
WHERE
    o.price > 45.0 OR o.customer_name = 'John'

	SELECT *
	FROM orders
	WHERE order_status = false
		AND region = 'beijing'
			AND name IN ('xiaoming', 'xiaohua')
) o

	OR o.customer_name = 'John'

  o.*,
  p.name,
  p.description,
  s.shipment_id,
  s.origin,
  s.destination,
  s.is_arrived
FROM
  orders AS o
  LEFT JOIN products AS p ON o.product_id=p.id
  LEFT JOIN shipments AS s ON o.order_id=s.order_id

	, s.destination, s.is_arrived

	LEFT JOIN shipments s ON o.order_id = s.order_id
