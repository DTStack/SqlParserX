 INSERT OVERWRITE TABLE  CMI.LOAN_ACCT_INFO_TEMP
     SELECT
         V_ETL_DATE AS DATE_STAT,
         A.ITEMNO   COD_IDSTR,
         A.ITEMNAME TXT_IDSTR,
         B.ITEMNO   COD_ONE_IDSTR,
         B.ITEMNAME TXT_ONE_IDSTR,
         C.ITEMNO   COD_TWO_IDSTR,
         C.ITEMNAME TXT_TWO_IDSTR,
         D.ITEMNO   COD_THREE_IDSTR,
         D.ITEMNAME TXT_THREE_IDSTR
     FROM OMI.CM_CODE_LIBRARY_HS A    --信贷参数表
     LEFT JOIN OMI.CM_CODE_LIBRARY_HS B
             ON B.ITEMNO = SUBSTR(A.ITEMNO,1,1)
             AND B.CODENO         = 'IndustryType'
             AND LENGTH(B.ITEMNO) = '1'
             AND B.BEGNDT <= v_etl_date
             AND B.OVERDT > v_etl_date
             AND B.PARTID = v_partid
     LEFT JOIN omi.cm_code_library_hs C
             ON C.ITEMNO = SUBSTR(A.ITEMNO,1,3)
             AND C.CODENO         = 'IndustryType'
             AND LENGTH(C.ITEMNO) = '3'
             AND C.BEGNDT <= v_etl_date
             AND C.OVERDT > v_etl_date
             AND C.PARTID = v_partid
     LEFT JOIN omi.cm_code_library_hs D
             ON D.ITEMNO = SUBSTR(A.ITEMNO,1,4)
             AND D.CODENO         = 'IndustryType'
             AND LENGTH(D.ITEMNO) = '4'
             AND D.BEGNDT <= v_etl_date
             AND D.OVERDT > v_etl_date
             AND D.PARTID = v_partid
     WHERE A.CODENO = 'IndustryType'
             AND LENGTH(A.ITEMNO) = '5'
             AND A.BEGNDT <= v_etl_date
             AND A.OVERDT > v_etl_date
             AND A.PARTID = v_partid;
