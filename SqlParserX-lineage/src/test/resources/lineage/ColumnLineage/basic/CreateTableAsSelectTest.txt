CREATE TABLE user_order_summary AS
SELECT
    u.id as user_id,
    u.name,
    SUM(o.amount) as total_amount,
    COUNT(o.id) as order_count
FROM users u JOIN orders o ON u.id = o.user_id
GROUP BY u.id, u.name

	'http://pdf.dfcfw.com/pdf/H6_',

	(
SELECT
  EID,
	OBJECTCODE,
	ADDRESS,
	EisDel,
	EITIME,
	EUTIME,
	row_number ( ) over ( PARTITION BY objectcode ORDER BY PICTURETYPE ) AS porder

	CDSY_PICTURES

	EDITTYPE NOT IN (101,301,307)
  AND EUTIME>=DATE_SUB(NOW(),INTERVAL 1 DAY)
	) A
WHERE
	A.porder = 1

CREATE TABLE tz
AS
SELECT EID AS ID, INFOCODE AS LAS_Code, AttachType AS AttachType, AttachName AS AttachName, AttachNo AS AttachNo
	, SUBSTRING_INDEX(ATTACHNAME, '/', -1) AS AttachOrigName
	, AttachSize AS AttachSize, AttachPages AS AttachPages, SOUATTACHNAME AS SouAttchname, 'H4' AS InfoType
	, SUBSTRING_INDEX(ATTACHNAME, SUBSTRING_INDEX(ATTACHNAME, '/', -1), 1) AS D_Path
	, CASE
		WHEN INSTR(ATTACHNAME, '.') > 0 THEN CONCAT(INFOCODE, '_', CAST(ROUND(ATTACHNO) AS CHAR), '.', SUBSTRING_INDEX(ATTACHNAME, '.', -1))
		ELSE CONCAT(INFOCODE, '_', CAST(ROUND(ATTACHNO) AS CHAR))
	END AS F_Name
	, CASE
		WHEN INSTR(ATTACHNAME, '.') > 0 THEN SUBSTRING_INDEX(ATTACHNAME, '.', -1)
		ELSE NULL
	END AS F_Type
	, CONCAT('http://pdf.dfcfw.com/pdf/H4_', INFOCODE, '_', CAST(ROUND(ATTACHNO) AS char), '.', SUBSTRING_INDEX(ATTACHNAME, '.', -1)) AS F_Url
	, EUTIME AS UpdateTime, EITIME AS EnterTime, 1 AS ResourceID, EISDEL AS EisDel
FROM INFO_NW_RELATTACH
WHERE EUTIME >= DATE_SUB(CURRENT_TIME(), INTERVAL 1 HOUR)

			when f.rat_score<4.5 and f.rat_score>=3.5 then '增持'

				sum(case when t.RATINGCHANGE not in (0,1,2) or t.RATINGCHANGE  is null then 1 else 0 end) as kp_num

create table tz as SELECT distinct
    a.SWDUCode  AS Union_SWCode,
    case when a.ColumnCode like '002001%' then substr(a.ColumnCode,1,9) else substr(a.ColumnCode,1,6) end as Xh_ColumnCode,
    t.ItemName AS RatingChangeName,
    NOW() AS EnterDBTime
FROM RPT_BaseInfo a
    INNER JOIN RPT_ReportBaseAdd h ON a.RPTCode = h.RPTCode
    LEFT JOIN RPT_RelateCode b ON a.RPTCode = b.RPTCode
    LEFT JOIN RPT_Encode t ON a.RatingChange = t.ItemCode
    LEFT JOIN RPT_Encode p ON a.AttachType = p.ItemCode
WHERE a.EisDel = '0'
    AND h.RECTime >= DATE_SUB( CURDATE(), INTERVAL 3 DAY )
GROUP BY a.RPTCode
    UNION ALL
SELECT DISTINCT
    a.PubCompanyName,
    a.INDSort  AS Union_SWName,
CASE
    WHEN c.SecuTypeCode = '058001001' THEN
    '01'
    WHEN c.SecuTypeCode = '058001002' THEN
    '02'
    WHEN c.SecuTypeCode LIKE '058001003%' THEN
    '03'
    WHEN c.SecuTypeCode = '058001008' THEN
    '08' ELSE NULL
    END AS SecuTypeCode,
    case when a.ColumnCode like '002001%' then substr(a.ColumnCode,1,9) else substr(a.ColumnCode,1,6) end as Xh_ColumnCode,
    NOW() AS EnterDBTime
FROM RPT_BaseInfo a
    INNER JOIN RPT_ReportBaseAdd h ON a.RPTCode = h.RPTCode
    LEFT JOIN RPT_RelateCode b ON a.RPTCode = b.RPTCode
    LEFT JOIN RPT_RelateCode c ON c.RPTCode = a.RPTCode
    LEFT JOIN RPT_Encode t ON a.RatingChange = t.ItemCode
    LEFT JOIN RPT_Encode p ON a.AttachType = p.ItemCode
WHERE
a.PubCompanyCode NOT IN ( '10014783', '10004823', '10983760' )
AND h.RECTime >= DATE_SUB(CURDATE(), INTERVAL 3 DAY )
GROUP BY a.RPTCode
