with tenant_users as (
    select id
    from user_info
    where tenant_id = ${cteTenant}
)
select u.id
from (
    select id, status
    from user_info
    where created_at >= ${startTime}
) u
join tenant_users tu on tu.id = u.id
where u.status in (${states})
  and exists (
      select 1
      from orders o
      where o.user_id = u.id
        and o.amount > ${minAmount}
  )
order by ${sortColumn} desc
limit ${offset}, ${rows}

with t1 as (
    select id, name, age
    from digua_ceshi.demo_table_digua
    where age >= ${t1MinAge}
),
t2 as (
    select sub.id, sub.name
    from (
        select id, name
        from digua_ceshi.demo_table_digua
        where name like ${t2Name}
    ) sub
),
t3 as (
    select id, name, age
    from digua_ceshi.new_table
    where age > ${t3AgeA}
    union all
    select id, name, age
    from digua_ceshi.new_table2
    where age <= ${t3AgeB}
),
t4 as (
    select t.id, t.name, nt.age as new_table_age
    from digua_ceshi.demo_table_digua t
    left join digua_ceshi.new_table nt
      on t.id = nt.id and nt.age > ${t4MinAge}
)
select t1.id,
       t1.name,
       t2.name as t2_name,
       t3.name as t3_name,
       t4.new_table_age as t4_age,
       dt.department,
       (select count(*)
        from digua_ceshi.demo_table_digua d
        where d.id = t1.id and d.age > ${olderAge}) as older_count
from t1
join t2 on t1.id = t2.id
left join t3 on t1.id = t3.id
left join t4 on t1.id = t4.id
join digua_ceshi.departments dt on t1.id = dt.employee_id
where dt.department = ${department}
union all
select t1.id,
       t1.name,
       t2.name as t2_name,
       t3.name as t3_name,
       t4.new_table_age as t4_age,
       dt.department,
       0 as older_count
from t1
join t2 on t1.id = t2.id
left join t3 on t1.id = t3.id
left join t4 on t1.id = t4.id
join digua_ceshi.departments dt on t1.id = dt.employee_id
where dt.department = ${backupDepartment}
